cmake_minimum_required(VERSION 3.24)
project(viper_action_contract LANGUAGES CXX)
add_library(viper_action_contract SHARED action_contract.cpp)
target_compile_features(viper_action_contract PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(viper_action_contract PRIVATE /W4 /fp:strict)
  set_target_properties(viper_action_contract PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
else()
  target_compile_options(viper_action_contract PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(action_contract_test action_contract_test.cpp)
  target_link_libraries(action_contract_test PRIVATE viper_action_contract)
  add_test(NAME action_contract COMMAND action_contract_test)
endif()
